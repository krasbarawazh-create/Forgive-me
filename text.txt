        // Floating Particle Generator
        function createFloatingElements() {
            const container = document.getElementById('heartsContainer');
            const icons = ['💖', '💍', '🫂', '🌸', '✨', '💕', '🎀'];
            for (let i = 0; i < 22; i++) {
                const el = document.createElement('div');
                el.className = 'floating-element text-lg sm:text-2xl';
                el.innerText = icons[Math.floor(Math.random() * icons.length)];
                el.style.left = Math.random() * 100 + 'vw';
                el.style.animationDelay = (Math.random() * 8) + 's';
                el.style.animationDuration = (6 + Math.random() * 6) + 's';
                container.appendChild(el);
            }
        }
        createFloatingElements();

        // Countdown Timer Logic
        function startCountdown() {
            const targetDate = new Date('2026-10-05T00:00:00').getTime();
            setInterval(() => {
                const now = new Date().getTime();
                const difference = targetDate - now;
                if (difference > 0) {
                    const days = Math.floor(difference / (1000 * 60 * 60 * 24));
                    const hours = Math.floor((difference % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                    const minutes = Math.floor((difference % (1000 * 60 * 60)) / (1000 * 60));
                    const seconds = Math.floor((difference % (1000 * 60)) / 1000);

                    document.getElementById('days').innerText = days < 10 ? '0' + days : days;
                    document.getElementById('hours').innerText = hours < 10 ? '0' + hours : hours;
                    document.getElementById('minutes').innerText = minutes < 10 ? '0' + minutes : minutes;
                    document.getElementById('seconds').innerText = seconds < 10 ? '0' + seconds : seconds;
                }
            }, 1000);
        }
        startCountdown();

        // Audio Context for Web Audio Synthesizer
        let audioCtx = null;
        function getAudioContext() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            return audioCtx;
        }

        let isBdayPlaying = false;

        function playPianoTone(freq, duration) {
            const ctx = getAudioContext();
            if (!ctx) return;
            try {
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(freq, ctx.currentTime);
                
                gain.gain.setValueAtTime(0.35, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration);

                osc.connect(gain);
                gain.connect(ctx.destination);
                osc.start();
                osc.stop(ctx.currentTime + duration);
            } catch(e) { console.log(e); }
        }

        function playBdayMelody() {
            stopAdeleAudio();
            stopBdayPiano();

            isBdayPlaying = true;
            document.getElementById('bdayText').innerText = "لە لێداندایە... 🎵";

            let step = 0;
            function nextNote() {
                if (!isBdayPlaying) return;
                if (step >= bdayNotes.length) {
                    step = 0;
                }
                const n = bdayNotes[step];
                playPianoTone(n.note, n.duration);
                step++;
                bdayTimeout = setTimeout(nextNote, n.duration * 550);
            }
            nextNote();
        }

        function stopBdayPiano() {
            isBdayPlaying = false;
            if (bdayTimeout) clearTimeout(bdayTimeout);
            document.getElementById('bdayText').innerText = "پیانۆی لەدایکبوون 🎂";
        }

        function toggleBdayPiano() {
            if (isBdayPlaying) {
                stopBdayPiano();
            } else {
                playBdayMelody();
            }
        }

        // --- Adele MP3 Audio Player ---
        const adeleAudio = document.getElementById('adeleAudio');
        let isAdelePlaying = false;

        function playAdeleAudio() {
            stopBdayPiano();
            adeleAudio.play().then(() => {
                isAdelePlaying = true;
                document.getElementById('romanticText').innerText = "لە لێداندایە... 🎶";
            }).catch(err => {
                console.log("Audio playback error:", err);
                // Fallback toggle state
                isAdelePlaying = false;
                document.getElementById('romanticText').innerText = "Lovesong - Adele 🎵";
            });
        }

        function stopAdeleAudio() {
            adeleAudio.pause();
            isAdelePlaying = false;
            document.getElementById('romanticText').innerText = "Lovesong - Adele 🎵";
        }

        function toggleAdeleAudio() {
            if (isAdelePlaying) {
                stopAdeleAudio();
            } else {
                playAdeleAudio();
            }
        }

        // Candle Blowing Action

                candleBlown = true;
                confetti({ particleCount: 160, spread: 100, origin: { y: 0.6 } });
                playBdayMelody();
            }
        }
